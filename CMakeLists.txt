cmake_minimum_required(VERSION 3.22)
project(LevelNativosAudio VERSION 0.4.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
if(APPLE)
  set(LEVEL_FORMATS VST3 AU Standalone)
else()
  set(LEVEL_FORMATS VST3 Standalone)
endif()
juce_add_plugin(Level COMPANY_NAME "Nativos Audio" BUNDLE_ID "com.nativosaudio.level"
  PLUGIN_MANUFACTURER_CODE NAtv PLUGIN_CODE LvL1 FORMATS ${LEVEL_FORMATS}
  PRODUCT_NAME "Level" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(Level)
target_sources(Level PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(Level PRIVATE Source)
target_compile_definitions(Level PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Level PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra
juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)

option(LEVEL_ENABLE_ONNX "Enable a licensed local ONNX Runtime build (requires ONNX models and static library)" OFF)
if(LEVEL_ENABLE_ONNX)
  if(NOT DEFINED ONNX_RUNTIME_INCLUDE_DIR OR NOT DEFINED ONNX_RUNTIME_STATIC_LIBRARY)
    message(FATAL_ERROR "Set ONNX_RUNTIME_INCLUDE_DIR and ONNX_RUNTIME_STATIC_LIBRARY for the selected platform. No models are bundled yet.")
  endif()
  target_include_directories(Level PRIVATE "${ONNX_RUNTIME_INCLUDE_DIR}")
  target_link_libraries(Level PRIVATE "${ONNX_RUNTIME_STATIC_LIBRARY}")
  target_compile_definitions(Level PRIVATE LEVEL_WITH_ONNX=1)
endif()

enable_testing()
add_executable(LevelRegionTests Tests/RegionTests.cpp)
target_compile_features(LevelRegionTests PRIVATE cxx_std_20)
target_include_directories(LevelRegionTests PRIVATE Source)
add_test(NAME LevelRegionTests COMMAND LevelRegionTests)

set(CPACK_PACKAGE_NAME "LEVEL-Nativos-Audio")
set(CPACK_PACKAGE_VENDOR "Nativos Audio")
set(CPACK_PACKAGE_VERSION "0.4.1")
set(CPACK_SOURCE_GENERATOR "ZIP")
include(CPack)
